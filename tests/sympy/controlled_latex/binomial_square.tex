\begin{theorem}[Binomial square]
For all real numbers $a$ and $b$,
\[
  (a+b)^2 = a^2 + 2ab + b^2.
\]
\end{theorem}

\begin{proof}
Expand the product $(a+b)(a+b)$ and collect like terms.
\end{proof}
