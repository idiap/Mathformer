\section*{Statement and Proof}

\begin{theorem}[Product-rule derivative]
For every real number $x$,
\[
  \frac{d}{dx}\left(x^2\sin x\right) = 2x\sin x + x^2\cos x.
\]
\end{theorem}

\begin{proof}
Apply the product rule to $x^2$ and $\sin x$.
\end{proof}
