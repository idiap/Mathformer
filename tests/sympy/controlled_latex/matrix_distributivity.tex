\section*{Statement and Proof}

\begin{theorem}[Matrix distributivity]
For matrices $A$, $B$, and $C$ with compatible dimensions,
\[
  A(B+C) = AB + AC.
\]
\end{theorem}

\begin{proof}
This is left distributivity of matrix multiplication over matrix addition.
\end{proof}
