\section*{Statement and Proof}

\begin{theorem}[Finite geometric sum]
For every natural number $n$ and every scalar $r$ with $r \ne 1$,
\[
  \sum_{k=0}^{n} r^k = \frac{1-r^{n+1}}{1-r}.
\]
\end{theorem}

\begin{proof}
Multiply the sum by $1-r$. The intermediate terms cancel, leaving
$1-r^{n+1}$, and division by $1-r$ is valid because $r \ne 1$.
\end{proof}
