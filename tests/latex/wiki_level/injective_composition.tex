\section*{Statement and Proof}

\begin{theorem}[Composition of injective functions]
Let $A$, $B$, and $C$ be sets, and let $f : A \to B$ and $g : B \to C$ be
functions. If $f$ and $g$ are injective, then the composite function
$g \circ f : A \to C$ is injective.
\end{theorem}

\begin{proof}
Let $x,y \in A$ and suppose that $(g \circ f)(x) = (g \circ f)(y)$.
By the definition of composition, this says $g(f(x)) = g(f(y))$. Since $g$ is
injective, it follows that $f(x) = f(y)$. Since $f$ is injective, it follows
that $x = y$. Therefore $g \circ f$ is injective.
\end{proof}
