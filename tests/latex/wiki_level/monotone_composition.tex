\section*{Controlled Origin Annotation}
This is a synthetic controlled case written in a Wikipedia-style exposition.
It is intended to test whether ordered structures are introduced at the right
level of generality and whether the target uses mathlib's \texttt{Monotone}.

\section*{Statement and Proof}

\begin{theorem}[Composition of monotone functions]
Let $A$, $B$, and $C$ be preordered sets. Let $f : A \to B$ and
$g : B \to C$ be monotone functions. Then $g \circ f : A \to C$ is monotone.
\end{theorem}

\begin{proof}
Let $x,y \in A$ and assume $x \le y$. Since $f$ is monotone, we have
$f(x) \le f(y)$. Since $g$ is monotone, applying $g$ to this inequality gives
$g(f(x)) \le g(f(y))$. This is exactly the statement that $g \circ f$ is
monotone.
\end{proof}
