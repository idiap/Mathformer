\section*{General Form}

\begin{theorem}[Hille--Yosida]
Let $X$ be a Banach space, let $A$ be a linear operator with domain
$D(A)\subseteq X$, let $\omega\in\mathbb{R}$, and let $M>0$.

Then $A$ generates a strongly continuous semigroup $(T(t))_{t\ge 0}$ satisfying
\[
\|T(t)\|\le M e^{\omega t}
\qquad\text{for all } t\ge 0,
\]
if and only if the following conditions hold:
\begin{enumerate}
    \item $A$ is closed and $D(A)$ is dense in $X$;
    \item every real $\lambda>\omega$ belongs to the resolvent set $\rho(A)$ of $A$, and for every such $\lambda$ and every positive integer $n$,
    \[
    \|(\lambda I-A)^{-n}\|
    \le
    \frac{M}{(\lambda-\omega)^n}.
    \]
\end{enumerate}
\end{theorem}

\section*{Contraction Semigroup Case}

\begin{theorem}[Hille--Yosida for contraction semigroups]
Let $X$ be a Banach space, and let $A$ be a linear operator with domain
$D(A)\subseteq X$.

Then $A$ generates a strongly continuous contraction semigroup
$(T(t))_{t\ge 0}$ if and only if:
\begin{enumerate}
    \item $A$ is closed and $D(A)$ is dense in $X$;
    \item every real $\lambda>0$ belongs to the resolvent set $\rho(A)$ of $A$, and
    \[
    \|(\lambda I-A)^{-1}\|\le \frac{1}{\lambda}.
    \]
\end{enumerate}
\end{theorem}

\section*{Remark}

A strongly continuous semigroup is also called a $C_0$-semigroup. The Hille--Yosida theorem characterizes precisely which closed densely defined operators generate such semigroups on Banach spaces.
