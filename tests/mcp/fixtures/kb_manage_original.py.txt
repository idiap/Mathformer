"""Manage the standalone Mathformer premise knowledge base."""

from __future__ import annotations

from typing import Any

import argparse
import json
from pathlib import Path
import sys

REPO_ROOT = Path(__file__).resolve().parents[1]
if str(REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(REPO_ROOT))

from backends.premise_kb.MathformerKB.indexing import (
    build_bloom_indexes,
    build_embedding_index,
)
from backends.premise_kb.MathformerKB.ingest import (
    entries_from_jsonl_corpus,
    entries_from_vampire_artifacts,
    generated_seed_entries,
    now,
)
from backends.premise_kb.MathformerKB.canonicalize import (
    stable_id_from_statement,
)
from backends.premise_kb.MathformerKB.schema import (
    KBEntry,
    ProvenanceRecord,
)
from backends.premise_kb.MathformerKB.search import (
    PremiseKB,
)
from backends.premise_kb.MathformerKB.storage import (
    KBStore,
)


def main() -> int:
    """Document the responsibility of ``main``.

    Motivation:
        Automate repository maintenance tasks that would be error-prone if performed by hand.

    Intent:
        Serve as the command-line entry point and coordinate argument parsing, execution, and reporting.

    Inputs:
        Accepts the parameters declared in the signature; callers should satisfy the invariants documented by neighboring code and tests.

    Outputs:
        Returns the value or side effect required by the workflow, with errors kept explicit at the call boundary.
    """
    argv = sys.argv[1:]
    json_requested = "--json" in argv
    argv = [item for item in argv if item != "--json"]
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--kb-root", type=Path, default=Path("artifacts/kb/premise_kb"))
    parser.add_argument("--json", action="store_true", default=json_requested)
    sub = parser.add_subparsers(dest="command", required=True)

    sub.add_parser("init")

    sub.add_parser("ingest-seeds")

    ingest_vampire = sub.add_parser("ingest-vampire")
    ingest_vampire.add_argument(
        "--artifact-root", type=Path, default=Path("artifacts/vampire_atp")
    )
    ingest_vampire.add_argument("--limit", type=int)

    ingest_corpus = sub.add_parser("ingest-corpus")
    ingest_corpus.add_argument("--path", type=Path, required=True)

    build_bloom = sub.add_parser("build-bloom")
    build_bloom.add_argument("--false-positive-rate", type=float, default=0.01)

    build_embeddings = sub.add_parser("build-embeddings")
    build_embeddings.add_argument("--dimension", type=int, default=512)

    sub.add_parser("refresh-canonical")

    has = sub.add_parser("has")
    has.add_argument("--surface", default="any")
    has.add_argument("--statement", required=True)

    search = sub.add_parser("search")
    search.add_argument("--query", required=True)
    search.add_argument("--k", type=int, default=10)
    search.add_argument("--domain-tag", action="append", default=[])
    search.add_argument("--proved-only", action="store_true")

    distance = sub.add_parser("distance")
    distance.add_argument("--query", required=True)
    distance.add_argument("--entry-id", required=True)
    distance.add_argument("--surface", default="best")

    enqueue = sub.add_parser("enqueue-source")
    enqueue.add_argument("--entry-file", type=Path, required=True)
    enqueue.add_argument("--reason", required=True)

    enqueue_web = sub.add_parser("enqueue-web")
    enqueue_web.add_argument("--url", required=True)
    enqueue_web.add_argument("--statement", required=True)
    enqueue_web.add_argument("--domain-tag", action="append", default=[])
    enqueue_web.add_argument(
        "--confidence", choices=["low", "medium", "high"], default="medium"
    )
    enqueue_web.add_argument("--reason", default="web enrichment candidate")

    promote = sub.add_parser("promote")
    promote.add_argument("--entry-id", required=True)
    promote.add_argument(
        "--status",
        choices=[
            "reviewed_informal",
            "formal_candidate",
            "proved_formal",
            "deprecated",
        ],
        required=True,
    )
    promote.add_argument("--notes")

    reject = sub.add_parser("reject")
    reject.add_argument("--entry-id", required=True)
    reject.add_argument("--notes")

    sub.add_parser("report")

    args = parser.parse_args(argv)
    kb_root = _repo_path(args.kb_root)
    store = KBStore(kb_root)

    if args.command == "init":
        store.init()
        payload = {"ok": True, "kb_root": str(kb_root)}
    elif args.command == "ingest-seeds":
        results = store.add_entries(generated_seed_entries())
        payload = {
            "ok": True,
            "results": [result.to_dict() for result in results],
            **store.report(),
        }
    elif args.command == "ingest-vampire":
        artifact_root = _repo_path(args.artifact_root)
        entries = entries_from_vampire_artifacts(artifact_root, limit=args.limit)
        results = store.add_entries(entries)
        payload = {
            "ok": True,
            "artifact_root": str(artifact_root),
            "candidate_count": len(entries),
            "results": [result.to_dict() for result in results],
            **store.report(),
        }
    elif args.command == "ingest-corpus":
        entries = entries_from_jsonl_corpus(_repo_path(args.path))
        results = store.add_entries(entries)
        payload = {
            "ok": True,
            "candidate_count": len(entries),
            "results": [r.to_dict() for r in results],
            **store.report(),
        }
    elif args.command == "build-bloom":
        paths = build_bloom_indexes(store, false_positive_rate=args.false_positive_rate)
        payload = {"ok": True, "paths": paths, **store.report()}
    elif args.command == "build-embeddings":
        path = build_embedding_index(store, dimension=args.dimension)
        payload = {"ok": True, "embedding_index": path, **store.report()}
    elif args.command == "refresh-canonical":
        refreshed = store.refresh_canonical_hashes()
        payload = {"ok": True, "refreshed": refreshed, **store.report()}
    elif args.command == "has":
        kb = PremiseKB(kb_root)
        payload = {
            "ok": True,
            "membership": kb.has_exact(args.statement, surface=args.surface).to_dict(),
        }
    elif args.command == "search":
        kb = PremiseKB(kb_root)
        hits = kb.search_semantic(
            args.query,
            k=args.k,
            domain_tags=args.domain_tag or None,
            proved_only=args.proved_only,
        )
        payload = {"ok": True, "query": args.query, "hits": hits}
    elif args.command == "distance":
        kb = PremiseKB(kb_root)
        payload = {
            "ok": True,
            "distance": kb.geometric_distance(
                args.query,
                entry_id=args.entry_id,
                surface=args.surface,
            ),
        }
    elif args.command == "enqueue-source":
        entry = KBEntry.from_dict(
            json.loads(_repo_path(args.entry_file).read_text(encoding="utf-8"))
        )
        store.enqueue_for_review(entry, reason=args.reason)
        payload = {"ok": True, "queued_entry_id": entry.id, "reason": args.reason}
    elif args.command == "enqueue-web":
        entry = KBEntry(
            id=stable_id_from_statement("web", args.statement),
            status="source_backed_candidate",
            domain_tags=list(args.domain_tag),
            informal_statement=args.statement,
            provenance=[
                ProvenanceRecord(
                    kind="web",
                    source=args.url,
                    retrieved_at=now(),
                    confidence=args.confidence,
                )
            ],
        )
        store.enqueue_for_review(entry, reason=args.reason)
        payload = {
            "ok": True,
            "queued_entry_id": entry.id,
            "source": args.url,
            "reason": args.reason,
        }
    elif args.command == "promote":
        provenance = ProvenanceRecord(
            kind="curation",
            source="scripts/manage_premise_kb.py promote",
            retrieved_at=now(),
            confidence="high",
            notes=args.notes,
        )
        entry = store.update_status(
            args.entry_id, args.status, provenance=provenance, notes=args.notes
        )
        payload = {"ok": True, "entry": entry.to_dict()}
    elif args.command == "reject":
        provenance = ProvenanceRecord(
            kind="curation",
            source="scripts/manage_premise_kb.py reject",
            retrieved_at=now(),
            confidence="high",
            notes=args.notes,
        )
        entry = store.update_status(
            args.entry_id, "rejected", provenance=provenance, notes=args.notes
        )
        payload = {"ok": True, "entry": entry.to_dict()}
    elif args.command == "report":
        payload = {"ok": True, **store.report()}
    else:
        raise AssertionError(args.command)

    if args.json:
        print(json.dumps(payload, ensure_ascii=False, indent=2))
    else:
        _print_human(payload)
    return 0 if payload.get("ok") else 3


def _repo_path(path: Path) -> Path:
    """Document the responsibility of ``_repo_path``.

    Motivation:
        Automate repository maintenance tasks that would be error-prone if performed by hand.

    Intent:
        Provide the local operation named ``_repo_path`` while preserving the surrounding module contract.

    Inputs:
        Accepts the parameters declared in the signature; callers should satisfy the invariants documented by neighboring code and tests.

    Outputs:
        Returns the value or side effect required by the workflow, with errors kept explicit at the call boundary.
    """
    return path if path.is_absolute() else REPO_ROOT / path


def _print_human(payload: dict[str, Any]) -> None:
    """Document the responsibility of ``_print_human``.

    Motivation:
        Automate repository maintenance tasks that would be error-prone if performed by hand.

    Intent:
        Provide the local operation named ``_print_human`` while preserving the surrounding module contract.

    Inputs:
        Accepts the parameters declared in the signature; callers should satisfy the invariants documented by neighboring code and tests.

    Outputs:
        Returns the value or side effect required by the workflow, with errors kept explicit at the call boundary.
    """
    print(f"ok: {payload.get('ok')}")
    for key in [
        "kb_root",
        "entry_count",
        "rejected_count",
        "candidate_count",
        "embedding_index",
        "queued_entry_id",
    ]:
        if key in payload:
            print(f"{key}: {payload[key]}")
    if "paths" in payload:
        print("paths:")
        for name, path in sorted(dict(payload["paths"]).items()):
            print(f"  {name}: {path}")
    if "membership" in payload:
        membership = dict(payload["membership"])
        print(f"surface: {membership['surface']}")
        print(f"probably_present: {membership['probably_present']}")
        if membership.get("matched_surfaces"):
            print(f"matched_surfaces: {', '.join(membership['matched_surfaces'])}")
        print(f"resolved_entry_ids: {', '.join(membership['resolved_entry_ids'])}")
    if "hits" in payload:
        print("hits:")
        for hit in payload["hits"]:
            entry = hit["entry"]
            print(
                f"  {hit['score']:.4f} dist={hit['cosine_distance']:.4f} "
                f"{entry['id']} [{entry['status']}]"
            )
    if "distance" in payload:
        distance = dict(payload["distance"])
        print(f"entry_id: {distance['entry_id']}")
        print(f"surface: {distance['surface']}")
        print(f"cosine_similarity: {distance['cosine_similarity']}")
        print(f"cosine_distance: {distance['cosine_distance']}")


if __name__ == "__main__":
    raise SystemExit(main())
