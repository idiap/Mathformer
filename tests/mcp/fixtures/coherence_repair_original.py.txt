"""Plan controlled MathformerPhysicsLib coherence repairs."""

from __future__ import annotations

from typing import Any

import argparse
import json
from pathlib import Path
import sys

SCRIPT_ROOT = (
    Path(__file__).resolve().parents[2]
    / "mathformerphysicslib-coherence-inventory-builder"
    / "scripts"
)
sys.path.insert(0, str(SCRIPT_ROOT))

from coherence_common import (  # noqa: E402 - direct-script path setup above
    audit_grounding,
    build_inventory,
    canonicalize,
    critic_api,
    critic_integration,
    critic_redefinitions,
    load_json,
    repair_plan,
    self_test_payload,
    write_json,
)


def default_reports(self_test: bool = False) -> list[dict[str, Any]]:
    inventory = self_test_payload() if self_test else build_inventory()
    graph = canonicalize(inventory)
    grounding = audit_grounding(graph)
    return [
        grounding,
        critic_redefinitions(graph, grounding),
        critic_integration(inventory, graph),
        critic_api(graph),
    ]


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--report", action="append", type=Path, default=[])
    parser.add_argument("--output", type=Path)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args(argv)

    reports = (
        [load_json(path) for path in args.report]
        if args.report
        else default_reports(args.self_test)
    )
    payload = repair_plan(reports)
    if args.output:
        write_json(payload, args.output)
    print(json.dumps(payload, indent=2, sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
