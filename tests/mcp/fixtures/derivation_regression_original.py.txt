"""Run regression tests for Lean physics derivation-completeness checks."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import subprocess
import sys
from typing import Any, Sequence

REPO_ROOT = Path(__file__).resolve().parents[2]
ARTIFACT_ROOT = (
    REPO_ROOT / "artifacts" / "physics_lean" / "derivation_completeness_regression"
)
FIXTURE_ROOT = REPO_ROOT / "tests" / "fixtures" / "physics_lean_adversarial"
CHECKER = "skills/physics-lean/physics-lean-derivation-completeness-checker/scripts/check_derivation_completeness.py"


def rel(path: Path) -> str:
    return path.relative_to(REPO_ROOT).as_posix()


def write_json(path: Path, payload: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )


def run_json(args: list[str]) -> dict[str, Any]:
    proc = subprocess.run(
        [sys.executable, *args],
        cwd=REPO_ROOT,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )
    if proc.returncode != 0:
        raise RuntimeError(
            f"command failed: {' '.join(args)}\nSTDOUT:\n{proc.stdout}\nSTDERR:\n{proc.stderr}"
        )
    return json.loads(proc.stdout)


def run_checker(name: str, *extra: str) -> dict[str, Any]:
    return run_json(
        [CHECKER, "--lean-file", rel(FIXTURE_ROOT / f"{name}.lean"), *extra, "--json"]
    )


def run_formalphysics_case(
    case_id: str, *, recursive_frontier_count: int | None = None
) -> dict[str, Any]:
    args = [
        CHECKER,
        "--lean-file",
        f"libraries/MathformerPhysicsLib/Lean/{case_id}/MathformerPhysics.lean",
        "--source-record",
        f"artifacts/physics_lean/dirac_recursive_batch/{case_id}/source_record.json",
    ]
    if recursive_frontier_count is None:
        args.extend(
            [
                "--recursive-plan",
                "artifacts/physics_lean/dirac_recursive_batch/recursive_grounding_plan.json",
            ]
        )
    else:
        args.extend(["--recursive-frontier-count", str(recursive_frontier_count)])
    args.append("--json")
    return run_json(args)


def codes(report: dict[str, Any]) -> set[str]:
    return {
        str(item.get("code"))
        for artifact in report.get("artifacts", [])
        for item in artifact.get("findings", [])
        if item.get("code")
    }


def first_artifact(report: dict[str, Any]) -> dict[str, Any]:
    return (report.get("artifacts") or [{}])[0]


def obligation_ids(report: dict[str, Any], *, blocked_only: bool = False) -> set[str]:
    artifact = first_artifact(report)
    obligations = (
        artifact.get("blocked_obligations")
        if blocked_only
        else artifact.get("obligations")
    )
    return {str(item.get("id")) for item in obligations or []}


def obligation_status(report: dict[str, Any], obligation_id: str) -> str:
    for item in first_artifact(report).get("obligations", []):
        if item.get("id") == obligation_id:
            return str(item.get("status"))
    return ""
