Only definitions are read. No registered action, skill, backend or scientific
workflow is executed. The generated corpus is a source-linked normalization

class DescriptorDumper(yaml.SafeDumper):
    """Emit independent JSON-compatible mappings, never YAML anchors/aliases."""

    def ignore_aliases(self, data):
        return True


def canonical_id(category, native):
    """Materialize an explicit, category-qualified alias mapping; collisions fail."""
    return category + '_' + re.sub(r'[^a-z0-9_]+', '_', native.lower()).strip('_') + '_workflow'


def source(path, selector):
    return {'path': path, 'selector': selector,
            'digest': 'sha256:' + hashlib.sha256((REPO_ROOT / path).read_bytes()).hexdigest()}


def owner_for_command(command):
    for token in command:
        path = Path(token)
        if path.parts[:1] == ('skills',) and len(path.parts) >= 4:
            owner = Path(*path.parts[:3]) / 'SKILL.md'
            if (REPO_ROOT / owner).is_file():
                return owner.as_posix()
    return DEFAULT_OWNER


def base(category, native, description, sources, owner, record, coverage):
    """Preserve the native contract alongside an intentionally scoped interface."""
    gaps = [
        'Native operation internals are not reimplemented as typed workflow nodes.',
        'Bindings name existing controller operations, discoverable and dispatchable through MCP; full workflow-skill runtime-policy qualification is not established by this projection.',
        'Workflow-level time/call limits are normalization bounds; legacy sources do not define equivalent global budgets.',
        'Network capabilities and complete external effects require native binding qualification; the hub safety label alone does not establish them.',
        'Receipts and mandatory provenance describe the v1 contract; existing optional native logs are not upgraded by normalization.',
    ] if coverage == 'interface_only' else [
        'This source is a declarative contract, not a callable workflow implementation.',
        'Operational order, branch guards, iteration state/bounds and typed invocation ports are not established by this source.',
        'The source contract is retained in mathformer.source; no executable implementation or output payload is invented.',
    ]
    return {
        'schema_version': 'workflow-skill/v1', 'namespace': 'mathformer',
        'id': canonical_id(category, native), 'version': '1.0.0', 'kind': 'workflow',
        'description': description, 'aliases': [native], 'lifecycle': 'draft',
        'activation': {'mode': 'explicit', 'prerequisites': []},
        'types': {'ContractIdentity': {'schema': {'type': 'string', 'const': native}, 'representation': 'value'}},
        'inputs': {}, 'outputs': {}, 'operations': {}, 'flow': {},
        'policy': {
            'effects': {'filesystem': {'read': ['repository'], 'write': []}, 'network': [], 'external_write': []},
            'authority': {'allowed': ['inspect_declared_contract'], 'forbidden': ['bypass_native_controls', 'infer_scientific_truth_from_completion']},
            'limits': {'wall_time_seconds': 3600, 'max_operation_calls': 1},
            'failure': {'default': 'fail', 'retry': {'max_attempts': 1, 'retryable_codes': []}},
            'resume': {'mode': 'none', 'require_matching_definition_and_inputs': True},
            'claim_ceiling': 'Source-linked workflow description only; no scientific result, runtime qualification or promotion is established.',
        },
        'observability': {'receipts': 'required', 'artifact_digests': 'sha256', 'provenance': 'required',
                          'events': ['workflow_started', 'step_started', 'step_completed', 'step_failed', 'workflow_completed']},
        'provenance': {'coverage': coverage, 'sources': sources, 'gaps': gaps},
        'extensions': {'mathformer.source': {'category': category, 'native_id': native,
                                           'owner_skill': owner, 'contract': record}},
    }


def native_interface(descriptor, actions, binding, extra_args=False):
    """Expose the real hub boundary, including dry-run/preflight/failure payloads."""
    descriptor['types'] = {
        'RunContext': {'schema': '../../hub-interface.schema.json#/$defs/RunContext', 'representation': 'value'},
        'ExtraArguments': {'schema': {'type': 'array', 'items': {'type': 'string'}}, 'representation': 'value'},
        'HubResult': {'schema': '../../hub-interface.schema.json#/$defs/HubResult', 'representation': 'value'},
    }
    ports = {'context': {'type': 'RunContext'}}
    arguments = {'context': {'ref': 'inputs.context'}}
    if extra_args:
        ports['extra_args'] = {'type': 'ExtraArguments'}
        arguments['extra_args'] = {'ref': 'inputs.extra_args'}
    descriptor['inputs'] = ports
    descriptor['outputs'] = {'result': {'type': 'HubResult', 'value': {'ref': 'steps.invoke.outputs.result'}}}
    descriptor['operations'] = {'invoke': {'inputs': ports, 'outputs': {'result': {'type': 'HubResult'}},
                                          'execution': {'mode': 'mixed', 'binding': binding}}}
    descriptor['flow'] = {'kind': 'sequence', 'steps': [
        {'id': 'invoke', 'kind': 'call', 'operation': 'invoke', 'arguments': arguments},
    ]}
    writes = set()
    visited = set()

    def visit(action_id):
        if action_id in visited:
            return
        visited.add(action_id)
        item = actions[action_id]
        writes.update(item.writes)
        for dependency in item.depends_on:
            visit(dependency)

    targets = descriptor['extensions']['mathformer.source']['contract'].get('action_ids', [binding['target']])
    for target in targets:
        visit(target)
    # Wrapper report writes are conditional on RunContext.write_report.
    writes.add('artifacts/workflow_hub/')
    descriptor['policy']['effects'] = {
        'filesystem': {'read': ['repository', 'native_runtime_inputs'], 'write': sorted(writes)},
        'network': ['native_binding_capabilities_require_qualification'],
        'external_write': ['native_binding_capabilities_require_qualification'],
    }
    descriptor['policy']['authority']['allowed'] = ['invoke_registered_native_workflow']
    descriptor['extensions']['mathformer.native_boundary'] = {
        'execution': 'Existing scripts/mathformer.py controller only; MCP can dispatch the canonical identity, but no general v1 DAG interpreter is supplied.',
        'context_fields': ['dry_run', 'json_output', 'keep_going', 'allow_mutating', 'write_report'],
        'effects_status': 'partial_native_declarations',
        'qualification_capability_names_are_not_runtime_grants': True,
        'failure_policy': 'Return the native ok/status payload unchanged; do not turn a failed subprocess or preflight into acceptance.',
        'scientific_payloads': 'HubResult describes the transport envelope, not backend reports or proof artifacts.',
    }


def declarative_interface(descriptor, reason):
    descriptor['operations'] = {'realize_contract': {
        'inputs': {}, 'outputs': {},
        'execution': {'mode': 'specification_only', 'reason': reason},
    }}
    descriptor['flow'] = {'kind': 'call', 'operation': 'realize_contract', 'arguments': {}}


def audit_action_dependencies(actions):
    """Reject unknown references and ancestor cycles before projection."""
    def check_graph(action_id, ancestors):
        if action_id not in actions:
            raise ValueError(f'unknown action dependency: {action_id}')
        if action_id in ancestors:
            raise ValueError(f'action dependency cycle: {action_id}')
        for child in actions[action_id].depends_on:
            check_graph(child, ancestors | {action_id})
    for action_id in actions:
        check_graph(action_id, set())

def render_descriptors(descriptors):
    """Serialize an already inventoried descriptor sequence without reordering."""
    files = {}
    records = []
    for descriptor in descriptors:
        native = descriptor['extensions']['mathformer.source']
        relative = Path(native['category']) / descriptor['id'] / 'workflow.yaml'
        files[relative] = HEADER + yaml.dump(descriptor, Dumper=DescriptorDumper, sort_keys=False, allow_unicode=True, width=100)
        records.append({'id': descriptor['id'], 'aliases': descriptor['aliases'], 'category': native['category'],
                        'path': relative.as_posix(), 'owner_skill': native['owner_skill'],
                        'description': descriptor['description'],
                        'descriptor_digest': 'sha256:' + hashlib.sha256(files[relative].encode()).hexdigest(),
                        'coverage': descriptor['provenance']['coverage'],
                        'binding_declared': any('binding' in op['execution'] for op in descriptor['operations'].values()),
                        'runtime_qualified': False, 'source_selectors': descriptor['provenance']['sources']})
    counts = {category: sum(row['category'] == category for row in records) for category in ('hub', 'route', 'loop', 'profile')}
    catalog = {'schema_version': 'mathformer.normalized-workflow-catalog/v1', 'descriptor_schema': '../workflow.schema.json',
               'generated_by': 'skills/control/workflow-contract-validator/scripts/normalize_workflows.py',
               'execution_migrated': False, 'counts': counts, 'workflows': records,
               'scope': 'All registered hub actions and NL routes, all hook loop contracts, and all canonical reasoning profiles. Ordinary skills remain skills; generated PRISM projections are not canonical workflow sources.'}
    files[Path('catalog.json')] = json.dumps(catalog, indent=2, ensure_ascii=False) + '\n'
    return files, catalog


def render():
    """Return deterministic file contents and a source-to-descriptor registry."""
    return render_descriptors(inventory())


def write_files(files, output=OUTPUT):
    """Write all files before validation, without rollback or deletion."""
    for relative, text in files.items():
        target = output / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(text, encoding='utf-8')


def check_rendered(files, catalog, output=OUTPUT):
    """Accumulate drift and validation errors in native traversal order."""
    errors = []
    for relative, text in files.items():
        target = output / relative
        if not target.is_file() or target.read_text(encoding='utf-8') != text:
            errors.append(f'missing or stale generated file: {target.relative_to(REPO_ROOT) if target.is_relative_to(REPO_ROOT) else target}')
            continue
        if relative.name == 'workflow.yaml':
            result = validate(target, REPO_ROOT)
            errors.extend(f'{relative}: {error}' for error in result['errors'])
    expected = {output / relative for relative in files}
    errors.extend(f'unexpected descriptor: {path}' for path in output.rglob('workflow.yaml') if path not in expected)
    return {'ok': not errors, 'counts': catalog['counts'], 'descriptor_count': len(catalog['workflows']),
            'schema_and_dataflow_validated': not errors, 'execution_migrated': False,
            'runtime_bindings_qualified': False, 'errors': errors}


def check(output=OUTPUT, write=False):
    """Check source coverage, drift, binding declarations and all descriptor contracts."""
    files, catalog = render()
    if write:
        write_files(files, output)
    return check_rendered(files, catalog, output)


def parse_options(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    modes = parser.add_mutually_exclusive_group(required=True)
    modes.add_argument('--check', action='store_true')
    modes.add_argument('--write', action='store_true')
    parser.add_argument('--json', action='store_true')
    args = parser.parse_args(argv)
    return {"write": args.write, "json": args.json}


def finish(result, options):
    if result is None and options.get('halted'):
        # argparse already emitted help and exited successfully before checking.
        return 0
    print(json.dumps(result, indent=2) if options["json"] else f'{"PASS" if result["ok"] else "FAIL"}: {result}')
    return 0 if result['ok'] else 1


def main(argv=None):
    options = parse_options(argv)
    return finish(check(write=options['write']), options)


if __name__ == '__main__':
    raise SystemExit(main())
