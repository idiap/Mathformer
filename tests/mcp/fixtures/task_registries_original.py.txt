# Motivation:
#   Keep Looper-style visible task registries connected to Mathformer's real
#   controlled hub, skill folders, validators, and artifacts.
#
# Intent:
#   Validate task registry JSON without adding runtime dependencies.
#
# Function:
#   Checks required fields, hub action references, skill references, artifact
#   states, validator declarations, generated-skill promotion boundaries, and
#   writes a structured report.
#
# Maintenance Notes:
#   Keep schema field names synchronized with meta_skills/schemas/task_registry.schema.json.

"""Validate Mathformer meta-skill task registries."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import sys
from typing import Any

REPO_ROOT = Path(__file__).resolve().parents[2]
SCRIPT_ROOT = REPO_ROOT / "scripts"
if str(SCRIPT_ROOT) not in sys.path:
    sys.path.insert(0, str(SCRIPT_ROOT))

import mathformer  # noqa: E402

SCHEMA_PATH = REPO_ROOT / "meta_skills" / "schemas" / "task_registry.schema.json"
REGISTRY_ROOT = REPO_ROOT / "meta_skills" / "registry" / "tasks"
OUTPUT_PATH = REPO_ROOT / "artifacts" / "meta_skills" / "task_registry_validation.json"


def rel(path: Path) -> str:
    try:
        return path.resolve().relative_to(REPO_ROOT).as_posix()
    except ValueError:
        return str(path)


def load_json(path: Path) -> dict[str, Any]:
    return json.loads(path.read_text(encoding="utf-8"))


def validate_registry(
    path: Path, schema: dict[str, Any], actions: set[str]
) -> list[dict[str, Any]]:
    data = load_json(path)
    findings: list[dict[str, Any]] = []
    required = schema["required"]
    for field in required:
        if field not in data:
            findings.append(
                {
                    "severity": "error",
                    "code": "missing_field",
                    "path": rel(path),
                    "field": field,
                }
            )

    if findings:
        return findings

    if data.get("schema_version") != "mathformer.meta_skills.task_registry.v0.1":
        findings.append(
            {
                "severity": "error",
                "code": "unexpected_schema_version",
                "path": rel(path),
            }
        )

    if path.parent.name != data.get("task_id"):
        findings.append(
            {
                "severity": "error",
                "code": "task_id_directory_mismatch",
                "path": rel(path),
                "task_id": data.get("task_id"),
                "directory": path.parent.name,
            }
        )

    nested = schema["nested_required"]
    for collection, fields in nested.items():
        rows = data.get(collection)
        if not isinstance(rows, list) or not rows:
            findings.append(
                {
                    "severity": "error",
                    "code": "empty_collection",
                    "path": rel(path),
                    "collection": collection,
                }
            )
            continue
        for index, row in enumerate(rows):
            if not isinstance(row, dict):
                findings.append(
                    {
                        "severity": "error",
                        "code": "non_object_row",
                        "path": rel(path),
                        "collection": collection,
                        "index": index,
                    }
                )
                continue
            for field in fields:
                if field not in row:
                    findings.append(
                        {
                            "severity": "error",
                            "code": "missing_nested_field",
                            "path": rel(path),
                            "collection": collection,
                            "index": index,
                            "field": field,
                        }
                    )

    for row in data.get("controlled_actions", []):
        action_id = row.get("action_id")
        if action_id not in actions:
            findings.append(
                {
                    "severity": "error",
                    "code": "unknown_action_id",
                    "path": rel(path),
                    "action_id": action_id,
                }
            )

    for workflow in data.get("visible_workflows", []):
        for action_id in workflow.get("entry_actions", []):
            if action_id not in actions:
                findings.append(
                    {
                        "severity": "error",
                        "code": "unknown_entry_action",
                        "path": rel(path),
                        "action_id": action_id,
                    }
                )

    for row in data.get("composed_skills", []):
        skill_id = row.get("skill_id")
        skill_paths = list((REPO_ROOT / "skills").glob(f"*/{skill_id}/SKILL.md"))
        if len(skill_paths) != 1:
            findings.append(
                {
                    "severity": "error",
                    "code": "missing_skill",
                    "path": rel(path),
                    "skill_id": skill_id,
                }
            )

    allowed_states = set(schema["allowed_artifact_states"])
    for row in data.get("artifacts", []):
        state = row.get("state")
        artifact_path = REPO_ROOT / str(row.get("path", ""))
        if state not in allowed_states:
            findings.append(
                {
                    "severity": "error",
                    "code": "invalid_artifact_state",
                    "path": rel(path),
                    "artifact": row,
                }
            )
        if state not in {"planned", "generated"} and not artifact_path.exists():
            findings.append(
                {
                    "severity": "error",
                    "code": "missing_artifact_path",
                    "path": rel(path),
                    "artifact": rel(artifact_path),
                }
            )

    for row in data.get("validators", []):
        command = str(row.get("command", ""))
        expected = REPO_ROOT / str(row.get("expected_artifact", ""))
        if not command.startswith("python "):
            findings.append(
                {
                    "severity": "error",
                    "code": "validator_not_python_command",
                    "path": rel(path),
                    "command": command,
                }
            )
        if not expected.as_posix().startswith((REPO_ROOT / "artifacts").as_posix()):
            findings.append(
                {
                    "severity": "error",
                    "code": "validator_expected_artifact_outside_artifacts",
                    "path": rel(path),
                    "expected_artifact": rel(expected),
                }
            )

    allowed_generated_states = set(schema["allowed_generated_skill_states"])
    for row in data.get("generated_skills", []):
        if row.get("state") not in allowed_generated_states:
            findings.append(
                {
                    "severity": "error",
                    "code": "invalid_generated_skill_state",
                    "path": rel(path),
                    "generated_skill": row,
                }
            )
        if not row.get("promotion_boundary"):
            findings.append(
                {
                    "severity": "error",
                    "code": "missing_generated_skill_promotion_boundary",
                    "path": rel(path),
                    "generated_skill": row,
                }
            )

    return findings


def run() -> dict[str, Any]:
    schema = load_json(SCHEMA_PATH)
    actions = set(mathformer.build_registry())
    registry_paths = sorted(REGISTRY_ROOT.glob("*/task_registry.json"))
    results = []
    all_findings = []
    for path in registry_paths:
        findings = validate_registry(path, schema, actions)
        all_findings.extend(findings)
        results.append(
            {
                "path": rel(path),
                "ok": not any(f["severity"] == "error" for f in findings),
                "findings": findings,
            }
        )
    payload = {
        "schema_version": 1,
        "style": "mathformer_meta_task_registry_validation",
        "ok": not any(f["severity"] == "error" for f in all_findings),
        "registry_count": len(results),
        "results": results,
        "findings": all_findings,
    }
    OUTPUT_PATH.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT_PATH.write_text(
        json.dumps(payload, indent=2, sort_keys=True), encoding="utf-8"
    )
    return payload


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()
    payload = run()
    if args.json:
        print(json.dumps(payload, indent=2, sort_keys=True))
    else:
        print("PASS task registries" if payload["ok"] else "FAIL task registries")
    return 0 if payload["ok"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
