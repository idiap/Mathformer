"""Run the Mathformer meta-skill validation suite."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import subprocess
import sys
import time
from typing import Any

REPO_ROOT = Path(__file__).resolve().parents[2]
PYTHON = sys.executable
OUTPUT_PATH = REPO_ROOT / "artifacts" / "meta_skills" / "meta_skill_suite.json"


COMMANDS = [
    ["meta_skills/scripts/validate_task_registries.py", "--json"],
    ["meta_skills/scripts/validate_skill_quality_cards.py", "--json"],
    ["meta_skills/scripts/record_skill_application.py", "--json"],
    ["meta_skills/scripts/validate_runtime_traces.py", "--json"],
    ["meta_skills/scripts/validate_workflow_evolution.py", "--json"],
    [
        "meta_skills/scripts/build_skill_projections.py",
        "--output",
        "artifacts/meta_skills/skill_projection_cards.json",
        "--report",
        "artifacts/meta_skills/skill_projection_report.json",
        "--json",
    ],
    ["meta_skills/scripts/navigate_workflow_taxonomy.py", "--self-test", "--json"],
    ["meta_skills/scripts/validate_generated_skill_lifecycle.py", "--json"],
    ["meta_skills/scripts/validate_mcp_boundary.py", "--json"],
    [
        "skills/physics-lean/physics-lean-goal-sample-autoformalizer/scripts/plan_goal_sample_autoformalization.py",
        "--case",
        "formalphysics_0014",
        "--json",
    ],
]


def run_command(parts: list[str]) -> dict[str, Any]:
    start = time.time()
    proc = subprocess.run(
        [PYTHON, *parts],
        cwd=REPO_ROOT,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=180,
    )
    duration = round(time.time() - start, 3)
    try:
        payload = json.loads(proc.stdout)
    except json.JSONDecodeError:
        payload = {"stdout": proc.stdout}
    return {
        "command": [PYTHON, *parts],
        "returncode": proc.returncode,
        "duration_seconds": duration,
        "ok": proc.returncode == 0 and payload.get("ok") is not False,
        "stdout": proc.stdout[-4000:],
        "stderr": proc.stderr[-4000:],
        "payload": payload,
    }


def run_suite() -> dict[str, Any]:
    results = [run_command(parts) for parts in COMMANDS]
    payload = {
        "schema_version": 1,
        "style": "mathformer_meta_skill_suite",
        "ok": all(row["ok"] for row in results),
        "check_count": len(results),
        "summary": {
            "pass": sum(1 for row in results if row["ok"]),
            "fail": sum(1 for row in results if not row["ok"]),
        },
        "results": results,
    }
    OUTPUT_PATH.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT_PATH.write_text(
        json.dumps(payload, indent=2, sort_keys=True), encoding="utf-8"
    )
    return payload


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()
    payload = run_suite()
    if args.json:
        print(json.dumps(payload, indent=2, sort_keys=True))
    else:
        print("PASS meta-skill suite" if payload["ok"] else "FAIL meta-skill suite")
    return 0 if payload["ok"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
