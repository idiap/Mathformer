def text(value: Any) -> str:
    """Escape a value for safe insertion into HTML text or attributes.

    Motivation:
        Reasoning logs may contain generated statements, paths, backend output,
        or model-facing summaries, so the renderer must never trust them as HTML.

    Intent:
        Convert arbitrary JSON-compatible values into escaped text.

    Inputs:
        Any value extracted from a reasoning-log event or validation report.

    Outputs:
        HTML-escaped text with empty fields normalized to the empty string.
    """

    if value is None:
        return ""
    return html.escape(str(value), quote=True)


def slug(value: Any) -> str:
    """Normalize a value for CSS classes and data attributes.

    Motivation:
        Filtering and status coloring need stable, simple tokens even when logs
        contain custom status strings.

    Intent:
        Preserve meaningful alphanumeric fragments and collapse everything else
        to hyphens.

    Inputs:
        A status, skill name, hook id, or step value.

    Outputs:
        A lowercase ASCII-ish slug suitable for CSS and data attributes.
    """

    raw = "" if value is None else str(value).strip().lower()
    return re.sub(r"[^a-z0-9]+", "-", raw).strip("-") or "unknown"


def compact_json(value: Any, *, limit: int = 1800) -> str:
    """Render JSON evidence compactly while keeping large payloads readable.

    Motivation:
        Raw event payloads are often useful, but massive one-line JSON blobs make
        a human dashboard hard to scan.

    Intent:
        Use pretty JSON, truncate only after serialization, and mark truncation.

    Inputs:
        A JSON-compatible value and a maximum character budget.

    Outputs:
        A readable JSON string or an empty string for empty values.
    """

    if value in ({}, [], None, ""):
        return ""
    rendered = json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True)
    if len(rendered) <= limit:
        return rendered
    half = max(100, limit // 2)
    return f"{rendered[:half]}\n...\n<truncated {len(rendered) - limit} characters>\n...\n{rendered[-half:]}"


def compact_inline(value: Any, *, limit: int = 500) -> str:
    """Render a value as one readable line for cards, rows, and search text."""
    if value in (None, "", {}, []):
        return ""
    if isinstance(value, (dict, list, tuple)):
        rendered = json.dumps(value, ensure_ascii=False, sort_keys=True)
    else:
        rendered = str(value)
    rendered = " ".join(rendered.split())
    if len(rendered) <= limit:
        return rendered
    return f"{rendered[: max(20, limit - 18)]} ... <truncated>"


def first_summary(values: Iterable[Any], *, limit: int = 500) -> str:
    """Return the first non-empty candidate as compact human-readable text."""
    for value in values:
        rendered = compact_inline(value, limit=limit)
        if rendered:
            return rendered
    return ""


def format_duration_ms(value: Any) -> str:
    """Format millisecond durations for human scanning."""
    if not isinstance(value, (int, float)):
        return ""
    if value < 1000:
        rendered = f"{value:.2f}".rstrip("0").rstrip(".")
        return f"{rendered} ms"
    seconds = value / 1000
    if seconds < 60:
        return f"{seconds:.2f} s"
    minutes = int(seconds // 60)
    remainder = seconds - minutes * 60
    return f"{minutes}m {remainder:.1f}s"


def rel(path: Path) -> str:
    """Return a repository-relative path when possible."""

    try:
        return str(path.resolve().relative_to(REPO_ROOT))
    except ValueError:
        return str(path)


def group_events_by_invocation(
    events: Sequence[Dict[str, Any]],
) -> Dict[str, List[Dict[str, Any]]]:
    """Group events by the canonical or compatibility skill-call id.

    Motivation:
        The dashboard should preserve the same grouping semantics as the
        Markdown renderer while still showing event-level detail.

    Intent:
        Reuse `event_skill_application` so new logs and legacy logs behave the
        same way.

    Inputs:
        Parsed reasoning-log event dictionaries.

    Outputs:
        A mapping from skill-call id to the events in that invocation group.
    """

    grouped: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
    for event in events:
        application = event_skill_application(event)
        call_id = str(application.get("skill_call_id") or "unknown-call")
        grouped[call_id].append(event)
    return dict(grouped)


def event_label(event: Mapping[str, Any]) -> str:
    """Build a short label for one event row."""

    event_index = event.get("event_index")
    step = event.get("step") or "unknown-step"
    return f"#{event_index} {step}" if event_index is not None else str(step)


def summarize_events(
    events: Sequence[Dict[str, Any]], validation: Optional[Dict[str, Any]]
) -> Dict[str, Any]:
    """Build dashboard-level counters and metadata.

    Motivation:
        Readers should know at a glance whether the trace is healthy, how large
        it is, which skills dominate it, and whether validation reported
        blockers.

    Intent:
        Derive small, high-signal counts without changing the underlying event
        records.

    Inputs:
        Parsed events and an optional validation report.

    Outputs:
        JSON-compatible summary fields consumed by the HTML renderer.
    """

    skills: Counter[str] = Counter()
    statuses: Counter[str] = Counter()
    hooks: Counter[str] = Counter()
    steps: Counter[str] = Counter()
    run_ids: Counter[str] = Counter()
    max_elapsed = 0.0

    for event in events:
        application = event_skill_application(event)
        skills[str(application.get("skill") or "unknown-skill")] += 1
        statuses[str(application.get("status") or "observed")] += 1
        hook_id = application.get("hook_id")
        if hook_id:
            hooks[str(hook_id)] += 1
        run_id = application.get("run_id")
        if run_id:
            run_ids[str(run_id)] += 1
        steps[str(event.get("step") or "unknown-step")] += 1
        elapsed = event.get("elapsed_ms_since_log_start")
        if isinstance(elapsed, (int, float)):
            max_elapsed = max(max_elapsed, float(elapsed))

    validation_summary = (validation or {}).get("summary", {})
    return {
        "event_count": len(events),
        "skill_count": len(skills),
        "step_count": len(steps),
        "hook_count": len(hooks),
        "run_count": len(run_ids),
        "max_elapsed_ms": round(max_elapsed, 2),
        "skills": dict(skills.most_common()),
        "statuses": dict(statuses.most_common()),
        "steps": dict(steps.most_common()),
        "hooks": dict(hooks.most_common()),
        "run_ids": dict(run_ids.most_common()),
        "validation_ok": None if validation is None else bool(validation.get("ok")),
        "finding_count": validation_summary.get("finding_count", 0),
        "blocking_findings": validation_summary.get("blocking_findings", 0),
    }


def render_stat(label: str, value: Any, note: str = "") -> str:
    """Render one dashboard statistic card."""

    note_html = f"<span>{text(note)}</span>" if note else ""
    return (
        '<div class="stat">'
        f"<strong>{text(value)}</strong>"
        f"<label>{text(label)}</label>"
        f"{note_html}"
        "</div>"
    )


def render_options(values: Iterable[str], *, default_label: str) -> str:
    """Render sorted `<option>` tags for a filter select."""

    lines = [f'<option value="">{text(default_label)}</option>']
    for value in sorted(values, key=lambda item: item.lower()):
        lines.append(f'<option value="{text(value)}">{text(value)}</option>')
    return "\n".join(lines)


def render_json_block(label: str, value: Any, *, open_by_default: bool = False) -> str:
    """Render a collapsible JSON block when useful evidence exists."""

    rendered = compact_json(value)
    if not rendered:
        return ""
    open_attr = " open" if open_by_default else ""
    return (
        f'<details class="json-block"{open_attr}>'
        f"<summary>{text(label)}</summary>"
        f"<pre>{text(rendered)}</pre>"
        "</details>"
    )


def render_invocation_card(
    invocation: Mapping[str, Any], events: Sequence[Dict[str, Any]]
) -> str:
    """Render one grouped skill invocation card.

    Motivation:
        The invocation is the level at which readers usually ask, "what did this
        skill try to do, what evidence did it produce, and what happened next?"

    Intent:
        Put the human summary first and keep raw details behind expandable
        sections.

    Inputs:
        A summary invocation from the Markdown renderer plus its raw events.

    Outputs:
        HTML for one filterable invocation card.
    """

    skill = str(invocation.get("skill") or "unknown-skill")
    status = str(invocation.get("status") or "observed")
    hook_id = str(invocation.get("hook_id") or "")
    call_id = str(invocation.get("skill_call_id") or "")
    is_legacy = call_id.startswith("legacy-")
    event_hook_ids: List[str] = []
    for event in events:
        event_hook_id = event_skill_application(event).get("hook_id")
        if event_hook_id and str(event_hook_id) not in event_hook_ids:
            event_hook_ids.append(str(event_hook_id))
    steps: List[str] = []
    for step in invocation.get("steps", []):
        step_text = str(step)
        if step_text not in steps:
            steps.append(step_text)
    searchable = " ".join(
        [
            skill,
            status,
            hook_id,
            " ".join(event_hook_ids),
            compact_inline(invocation.get("intent")),
            compact_inline(invocation.get("human_summary")),
            " ".join(steps),
            compact_inline(invocation.get("evidence"), limit=600),
            compact_inline(invocation.get("decision"), limit=600),
            compact_inline(invocation.get("risks"), limit=600),
        ]
    )

    event_items = []
    for event in events:
        application = event_skill_application(event)
        payload = event.get("payload")
        if not isinstance(payload, dict):
            payload = {}
        summary = first_summary(
            [
                application.get("human_summary"),
                payload.get("human_summary"),
                payload.get("summary"),
                payload.get("reason"),
            ]
        )
        elapsed = event.get("elapsed_ms_since_log_start")
        elapsed_text = format_duration_ms(elapsed)
        event_items.append(
            "<li>"
            f"<span>{text(event_label(event))}</span>"
            f"<em>{text(elapsed_text)}</em>"
            f"<p>{text(summary)}</p>"
            f"{render_json_block('event payload', payload)}"
            "</li>"
        )

    badges = [
        f'<span class="badge status-{text(slug(status))}">{text(status)}</span>',
        f'<span class="badge">{text(invocation.get("event_count"))} events</span>',
    ]
    for visible_hook_id in event_hook_ids[:3]:
        badges.append(f'<span class="badge hook">{text(visible_hook_id)}</span>')
    if len(event_hook_ids) > 3:
        badges.append(
            f'<span class="badge hook">+{text(len(event_hook_ids) - 3)} hooks</span>'
        )
    if is_legacy:
        badges.append('<span class="badge legacy">legacy log</span>')

    intent = compact_inline(invocation.get("intent"))
    if not intent and is_legacy:
        intent = "Legacy hook trace; no skill-application intent was recorded."
    elif not intent:
        intent = "Intent not recorded."

    return (
        '<article class="invocation-card" '
        f'data-skill="{text(skill)}" '
        f'data-status="{text(status)}" '
        f'data-steps="{text(" ".join(steps))}" '
        f'data-search="{text(searchable.lower())}">'
        '<div class="card-topline">'
        f"<h3>{text(skill)}</h3>"
        f"<div>{''.join(badges)}</div>"
        "</div>"
        f'<p class="intent">{text(intent)}</p>'
        f"<p>{text(compact_inline(invocation.get('human_summary')))}</p>"
        '<dl class="meta-grid">'
        f"<dt>skill call</dt><dd>{text(call_id)}</dd>"
        f"<dt>run</dt><dd>{text(invocation.get('run_id'))}</dd>"
        f"<dt>next</dt><dd>{text(invocation.get('next_action'))}</dd>"
        "</dl>"
        '<div class="evidence-grid">'
        f"{render_json_block('inputs', invocation.get('inputs'))}"
        f"{render_json_block('artifacts', invocation.get('artifacts'), open_by_default=True)}"
        f"{render_json_block('evidence', invocation.get('evidence'), open_by_default=True)}"
        f"{render_json_block('decision', invocation.get('decision'))}"
        f"{render_json_block('risks', invocation.get('risks'))}"
        "</div>"
        '<details class="event-list">'
        f"<summary>Events in this invocation ({text(len(events))})</summary>"
        f"<ol>{''.join(event_items)}</ol>"
        "</details>"
        "</article>"
    )


def render_timeline_rows(events: Sequence[Dict[str, Any]]) -> str:
    """Render the event-level timeline table body."""

    rows = []
    for event in events:
        application = event_skill_application(event)
        payload = event.get("payload")
        if not isinstance(payload, dict):
            payload = {}
        status = str(application.get("status") or "observed")
        skill = str(application.get("skill") or "unknown-skill")
        step = str(event.get("step") or "unknown-step")
        summary = first_summary(
            [
                application.get("human_summary"),
                payload.get("human_summary"),
                payload.get("summary"),
                payload.get("reason"),
            ]
        )
        elapsed = event.get("elapsed_ms_since_log_start")
        elapsed_text = format_duration_ms(elapsed)
        rows.append(
            '<tr class="timeline-row" '
            f'data-skill="{text(skill)}" '
            f'data-status="{text(status)}" '
            f'data-steps="{text(step)}" '
            f'data-search="{text((skill + " " + status + " " + step + " " + summary).lower())}">'
            f"<td>{text(event.get('event_index'))}</td>"
            f"<td>{text(elapsed_text)}</td>"
            f"<td>{text(skill)}</td>"
            f"<td>{text(step)}</td>"
            f"<td>{text(status)}</td>"
            f"<td>{text(summary)}</td>"
            "</tr>"
        )
    return "\n".join(rows)


def render_findings(validation: Optional[Dict[str, Any]]) -> str:
    """Render validation findings, or a concise validation status when clean."""

    if validation is None:
        return (
            '<section class="panel">'
            "<h2>Validation</h2>"
            "<p>Validation was not run for this dashboard.</p>"
            "</section>"
        )

    findings = validation.get("findings", [])
    status = "passed" if validation.get("ok") else "reported blockers"
    if not findings:
        return (
            '<section class="panel validation-ok">'
            "<h2>Validation</h2>"
            f"<p>Schema validation {text(status)} with no findings.</p>"
            "</section>"
        )

    rows = []
    for item in findings:
        rows.append(
            "<tr>"
            f"<td>{text(item.get('severity'))}</td>"
            f"<td>{text(item.get('code'))}</td>"
            f"<td>{text(item.get('event_index'))}</td>"
            f"<td>{text(item.get('step'))}</td>"
            f"<td>{text(item.get('message'))}</td>"
            "</tr>"
        )
    return (
        '<section class="panel validation-warn">'
        "<h2>Validation Findings</h2>"
        f"<p>{text(len(findings))} finding(s). Blocking: {text(validation.get('summary', {}).get('blocking_findings', 0))}.</p>"
        '<div class="table-wrap"><table>'
        "<thead><tr><th>Severity</th><th>Code</th><th>Event</th><th>Step</th><th>Message</th></tr></thead>"
        f"<tbody>{''.join(rows)}</tbody>"
        "</table></div>"
        "</section>"
    )

def default_hooks_file() -> Optional[Path]:
    """Return the repository hook contract when it is available."""

    hooks_file = REPO_ROOT / "skills" / "SKILL_HOOKS.yaml"
    return hooks_file if hooks_file.exists() else None


def maybe_validate(
    args: argparse.Namespace, log_file: Path
) -> Optional[Dict[str, Any]]:
    """Run optional schema validation for the dashboard.

    Motivation:
        A friendly view should also tell the reader whether the underlying trace
        passed the formal log contract.

    Intent:
        Validate by default when a hook file is available, while allowing callers
        to skip validation for partial or external logs.

    Inputs:
        Parsed CLI arguments and the log path.

    Outputs:
        A validation report dictionary, or `None` when validation is disabled.
    """

    if args.no_validation:
        return None
    hooks_file = (
        args.hooks_file if args.hooks_file is not None else default_hooks_file()
    )
    return validate_reasoning_log(
        log_file,
        hooks_file=hooks_file,
        strict=args.strict,
        require_instrumentation=args.require_instrumentation,
        require_skill_application=args.require_skill_application,
    )


def parse_args(argv: Optional[Sequence[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments for the HTML renderer."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--log-file",
        type=Path,
        help="Input reasoning-log JSONL file. Defaults to artifacts/reasoning/self_test/good.jsonl.",
    )
    parser.add_argument(
        "--output-file",
        type=Path,
        help="Output HTML file. Defaults to the log path with .html suffix.",
    )
    parser.add_argument(
        "--title", default="Mathformer Reasoning Log", help="Dashboard title."
    )
    parser.add_argument(
        "--hooks-file", type=Path, help="Optional SKILL_HOOKS.yaml path for validation."
    )
    parser.add_argument(
        "--strict",
        action="store_true",
        help="Run strict schema validation before rendering.",
    )
    parser.add_argument(
        "--require-instrumentation",
        action="store_true",
        help="Require agent_usage and convergence metadata.",
    )
    parser.add_argument(
        "--require-skill-application",
        action="store_true",
        help="Require payload.skill_application on every event.",
    )
    parser.add_argument(
        "--no-validation",
        action="store_true",
        help="Render without running schema validation.",
    )
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Render the requested reasoning log to HTML."""

    args = parse_args(argv)
    log_file = args.log_file or Path("artifacts/reasoning/self_test/good.jsonl")
    if not log_file.exists():
        from audit_reasoning_log_quality import write_self_test_fixtures

        write_self_test_fixtures()
    log_file = log_file.resolve()
    output_file = args.output_file or log_file.with_suffix(".html")
    events = load_events(log_file)
    validation = maybe_validate(args, log_file)
    html_text = render_html_dashboard(
        log_file=log_file,
        events=events,
        validation=validation,
        title=args.title,
    )
    output_file.parent.mkdir(parents=True, exist_ok=True)
    output_file.write_text(html_text, encoding="utf-8")
    print(str(output_file))
    if validation is not None and validation.get("summary", {}).get(
        "blocking_findings", 0
    ):
        return 2
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
