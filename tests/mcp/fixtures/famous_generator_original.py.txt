def main() -> int:
    """Document the responsibility of ``main``.

    Motivation:
        Make agent behavior explicit, scoped, and reproducible for specialized Mathformer tasks.

    Intent:
        Serve as the command-line entry point and coordinate argument parsing, execution, and reporting.

    Inputs:
        Accepts the parameters declared in the signature; callers should satisfy the invariants documented by neighboring code and tests.

    Outputs:
        Returns the value or side effect required by the workflow, with errors kept explicit at the call boundary.
    """
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--output",
        type=Path,
        default=Path("artifacts/kb/famous_theorems_axioms_500.jsonl"),
    )
    parser.add_argument("--limit", type=int, default=500)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()

    entries = build_entries(limit=args.limit)
    output = args.output if args.output.is_absolute() else REPO_ROOT / args.output
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(
        "".join(
            json.dumps(entry.to_dict(), ensure_ascii=False, sort_keys=True) + "\n"
            for entry in entries
        ),
        encoding="utf-8",
    )
    payload = {
        "ok": True,
        "output": str(output),
        "entry_count": len(entries),
        "expected_count": args.limit,
        "domain_count": len(DOMAIN_CATALOG),
        "entry_ids": [entry.id for entry in entries],
    }
    if args.json:
        print(json.dumps(payload, ensure_ascii=False, indent=2))
    else:
        print(f"ok: {payload['ok']}")
        print(f"output: {payload['output']}")
        print(f"entry_count: {payload['entry_count']}")
    return 0 if len(entries) == args.limit else 3


def build_entries(*, limit: int) -> list[KBEntry]:
    """Document the responsibility of ``build_entries``.

    Motivation:
        Make agent behavior explicit, scoped, and reproducible for specialized Mathformer tasks.

    Intent:
        Build the local data structure or artifact used by the next workflow stage.

    Inputs:
        Accepts the parameters declared in the signature; callers should satisfy the invariants documented by neighboring code and tests.

    Outputs:
        Returns the value or side effect required by the workflow, with errors kept explicit at the call boundary.
    """
    entries: list[KBEntry] = []
    for domain_id, payload in DOMAIN_CATALOG.items():
        names = list(payload["names"])  # type: ignore[arg-type]
        if len(names) != 20:
            raise ValueError(
                f"Domain {domain_id!r} must have exactly 20 names; got {len(names)}."
            )
        for name in names:
            entries.append(make_entry(domain_id, payload, str(name)))
    if len(entries) < limit:
        raise ValueError(f"Catalog has only {len(entries)} entries; requested {limit}.")
    return entries[:limit]
