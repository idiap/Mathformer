from __future__ import annotations

import argparse
import json
from pathlib import Path
from typing import Any

REPO_ROOT = Path(__file__).resolve().parents[3]


def repo_path(path: str) -> Path:
    """Return an absolute repository path for a repo-relative string."""

    return REPO_ROOT / path


def path_entry(path: str) -> dict[str, Any]:
    """Create a path record with existence metadata.

    Motivation:
        Paper claims should know whether a cited implementation or artifact is
        present in the current checkout.

    Intent:
        Store both the display path and a lightweight status flag.
    """

    candidate = repo_path(path)
    return {
        "path": path,
        "exists": candidate.exists(),
        "kind": (
            "directory"
            if candidate.is_dir()
            else "file" if candidate.exists() else "missing"
        ),
    }


def validation_command(
    command: str, output_paths: list[str] | None = None
) -> dict[str, Any]:
    """Create a validation command record without executing it."""

    return {
        "command": command,
        "output_paths": [path_entry(path) for path in (output_paths or [])],
    }

FEATURE_BUILDERS = {
    "reasoning-log-html-renderer": reasoning_log_renderer_packet,
    "reference-free-critic-monitor": reference_free_monitor_packet,
    "controlled-action-hub": controlled_hub_packet,
    "workflow-catalogue-graph-report": workflow_catalog_packet,
}


def build_packet(feature: str) -> dict[str, Any]:
    """Build one feature packet or a combined packet."""

    if feature == "all":
        packets = [builder() for builder in FEATURE_BUILDERS.values()]
        return {
            "schema_version": 1,
            "packet_type": "paper_new_functionality_evidence_collection",
            "features": packets,
            "feature_count": len(packets),
            "missing_paths": missing_paths(packets),
        }
    packet = FEATURE_BUILDERS[feature]()
    packet["schema_version"] = 1
    packet["packet_type"] = "paper_new_functionality_evidence"
    packet["missing_paths"] = missing_paths([packet])
    return packet


def iter_path_records(value: Any) -> list[dict[str, Any]]:
    """Collect nested path records from a packet."""

    records: list[dict[str, Any]] = []
    if isinstance(value, dict):
        if {"path", "exists", "kind"}.issubset(value):
            records.append(value)
        for child in value.values():
            records.extend(iter_path_records(child))
    elif isinstance(value, list):
        for child in value:
            records.extend(iter_path_records(child))
    return records


def missing_paths(packets: list[dict[str, Any]]) -> list[str]:
    """Return all packet paths that do not exist in the checkout."""

    missing: list[str] = []
    for packet in packets:
        for record in iter_path_records(packet):
            if not record["exists"]:
                missing.append(record["path"])
    return sorted(set(missing))


def main() -> int:
    """Parse arguments, build a packet, and optionally emit JSON output."""

    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--feature",
        choices=["all", *sorted(FEATURE_BUILDERS)],
        default="all",
        help="Feature packet to build.",
    )
    parser.add_argument(
        "--output",
        default="artifacts/paper_skills/new_functionality_evidence_packet.json",
        help="Repository-relative or absolute output JSON path.",
    )
    parser.add_argument("--json", action="store_true", help="Print a JSON summary.")
    args = parser.parse_args()

    packet = build_packet(args.feature)
    output = Path(args.output)
    if not output.is_absolute():
        output = REPO_ROOT / output
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(
        json.dumps(packet, indent=2, sort_keys=True) + "\n", encoding="utf-8"
    )

    result = {
        "ok": True,
        "output": str(output),
        "feature": args.feature,
        "feature_count": packet.get("feature_count", 1),
        "missing_path_count": len(packet.get("missing_paths", [])),
        "missing_paths": packet.get("missing_paths", []),
    }
    if args.json:
        print(json.dumps(result, indent=2, sort_keys=True))
    else:
        print(f"Wrote {output}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
