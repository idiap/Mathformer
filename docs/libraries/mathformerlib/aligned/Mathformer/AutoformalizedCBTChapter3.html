def inclusionEmbedding {α : Type u} {A B : Set α} (hAB : A ⊆ B) : A ↪ B where

  inj' := by
    intro x y hxy
    cases x with
    | mk xv hx =>
      cases y with
      | mk yv hy =>
        simp at hxy
        simp [hxy]

theorem samePower_of_subset_sandwich {α : Type u} {A B C : Set α}
    (hCB : C ⊆ B) (hBA : B ⊆ A) (hAC : SamePower A C) :
    SamePower A B := by
